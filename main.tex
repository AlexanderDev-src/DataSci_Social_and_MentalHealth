\setlist[subitems]{label=\thesection.\arabic*,labelindent=1.27cm,leftmargin=*,

% รายการที่อยู่ใต้ \subsection -- ต้องใช้เลข 3 ระดับ ไม่งั้นชนกับเลขหัวข้อย่อย
